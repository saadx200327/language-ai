rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function signedIn() {
      return request.auth != null;
    }

    function isOwner(userId) {
      return signedIn() && request.auth.uid == userId;
    }

    match /users/{userId} {
      allow read, create, update: if isOwner(userId);
      allow delete: if false;
    }

    match /leaderboard/{userId} {
      allow read: if true;
      allow create, update: if isOwner(userId)
        && request.resource.data.uid == request.auth.uid
        && request.resource.data.keys().hasOnly(['uid','name','xp','streak','badge','courseId','updatedAt'])
        && request.resource.data.name is string
        && request.resource.data.name.size() <= 24
        && request.resource.data.xp is number
        && request.resource.data.xp >= 0
        && request.resource.data.streak is number
        && request.resource.data.streak >= 0
        && request.resource.data.courseId in ['bn_en','en_es'];
      allow delete: if false;
    }
  }
}
